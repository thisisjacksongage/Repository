<script>

const STORAGE_KEY = "moneytrack_data_v1";

let data = {
  goals: [
    {
      id: "graduation",
      name: "Graduation Savings",
      target: 10000,
      amount: 900
    },
    {
      id: "laptop",
      name: "Laptop",
      target: 1200,
      amount: 0
    },
    {
      id: "printer",
      name: "3D Printer",
      target: 400,
      amount: 0
    }
  ],

  transactions: [],

  depop: []
};


// -------------------------
// LOAD / SAVE
// -------------------------

function loadData() {

  const saved = localStorage.getItem(STORAGE_KEY);

  if (saved) {

    try {
      data = JSON.parse(saved);
    } catch {
      console.log("Could not load saved data.");
    }

  }

}


function saveData() {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
}


// -------------------------
// HELPERS
// -------------------------

function money(value) {

  return new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: "USD"
  }).format(value || 0);

}


function today() {

  return new Date().toISOString().split("T")[0];

}


function currentMonth() {

  return new Date().toISOString().slice(0, 7);

}


function escapeHTML(str) {

  return String(str || "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");

}


// -------------------------
// GOALS
// -------------------------

function renderGoals() {

  const container = document.getElementById("goals");

  container.innerHTML = "";

  data.goals.forEach(goal => {

    const percent = Math.min(
      100,
      (goal.amount / goal.target) * 100
    );

    const div = document.createElement("div");

    div.className = "goal";

    div.innerHTML = `

      <div class="goal-top">

        <div class="goal-name">
          ${escapeHTML(goal.name)}
        </div>

        <div class="goal-amount">
          ${money(goal.amount)} / ${money(goal.target)}
        </div>

      </div>

      <div class="progress">

        <div
          class="progress-bar ${percent >= 100 ? "complete" : ""}"
          style="width:${percent}%"
        ></div>

      </div>

      <div
        class="muted"
        style="font-size:12px;margin-top:5px;"
      >
        ${percent.toFixed(1)}%
      </div>

    `;

    container.appendChild(div);

  });

}


function renderGoalSelect() {

  const select = document.getElementById("moneyGoal");

  select.innerHTML = "";

  data.goals.forEach(goal => {

    const option = document.createElement("option");

    option.value = goal.id;

    option.textContent = goal.name;

    select.appendChild(option);

  });

}


// -------------------------
// STATS
// -------------------------

function updateStats() {

  const graduation =
    data.goals.find(g => g.id === "graduation");

  const totalMoney =
    graduation ? graduation.amount : 0;

  const totalProfit =
    data.depop.reduce(
      (sum, sale) => sum + sale.profit,
      0
    );

  const month = currentMonth();

  const monthTransactions =
    data.transactions.filter(t =>
      t.date.startsWith(month)
    );

  const income =
    monthTransactions
      .filter(t => t.type === "income")
      .reduce((sum, t) => sum + t.amount, 0);

  const spending =
    monthTransactions
      .filter(t => t.type === "expense")
      .reduce((sum, t) => sum + t.amount, 0);

  const monthProfit =
    data.depop
      .filter(s => s.date.startsWith(month))
      .reduce((sum, s) => sum + s.profit, 0);

  document.getElementById("totalMoney").textContent =
    money(totalMoney);

  document.getElementById("graduationProgress").textContent =
    money(totalMoney);

  document.getElementById("totalProfit").textContent =
    money(totalProfit);

  document.getElementById("monthlyNet").textContent =
    money(income - spending + monthProfit);

  document.getElementById("monthIncome").textContent =
    money(income);

  document.getElementById("monthSpending").textContent =
    money(spending);

  document.getElementById("monthDepopProfit").textContent =
    money(monthProfit);

}


// -------------------------
// ACTIVITY
// -------------------------

function renderActivity() {

  const table =
    document.getElementById("activityTable");

  table.innerHTML = "";

  const activities = [];

  data.transactions.forEach(t => {

    activities.push({
      id: t.id,
      date: t.date,
      type: t.type,
      description: t.note || "Transaction",
      goal: getGoalName(t.goalId),
      amount:
        t.type === "expense"
          ? -t.amount
          : t.amount,
      profit: null
    });

  });


  data.depop.forEach(s => {

    activities.push({
      id: s.id,
      date: s.date,
      type: "Depop Sale",
      description: s.item,
      goal: s.source || "-",
      amount: s.sale,
      profit: s.profit,
      depop: true
    });

  });


  activities.sort((a,b) =>
    new Date(b.date) - new Date(a.date)
  );


  if (activities.length === 0) {

    table.innerHTML = `
      <tr>
        <td colspan="7" class="empty">
          No activity yet.
        </td>
      </tr>
    `;

    return;

  }


  activities.forEach(a => {

    const row =
      document.createElement("tr");

    const amountClass =
      a.amount >= 0
        ? "positive"
        : "negative";

    row.innerHTML = `

      <td>${escapeHTML(a.date)}</td>

      <td>${escapeHTML(a.type)}</td>

      <td>${escapeHTML(a.description)}</td>

      <td>${escapeHTML(a.goal)}</td>

      <td class="${amountClass}">
        ${money(a.amount)}
      </td>

      <td>
        ${
          a.profit !== null
            ? `<span class="${a.profit >= 0 ? "positive" : "negative"}">
                ${money(a.profit)}
              </span>`
            : "-"
        }
      </td>

      <td>
        <button
          class="danger"
          style="padding:6px 9px;width:auto;"
          onclick="deleteActivity('${a.id}', ${a.depop ? "true" : "false"})"
        >
          Delete
        </button>
      </td>

    `;

    table.appendChild(row);

  });

}


function getGoalName(id) {

  const goal =
    data.goals.find(g => g.id === id);

  return goal ? goal.name : "-";

}


// -------------------------
// ADD GOAL
// -------------------------

document
  .getElementById("goalForm")
  .addEventListener("submit", function(e) {

    e.preventDefault();

    const name =
      document.getElementById("goalName").value.trim();

    const target =
      Number(document.getElementById("goalTarget").value);

    const starting =
      Number(document.getElementById("goalStarting").value) || 0;

    if (!name || target <= 0) {
      alert("Enter a valid goal.");
      return;
    }

    data.goals.push({

      id:
        "goal_" +
        Date.now(),

      name,
      target,
      amount: starting

    });

    saveData();

    this.reset();

    document.getElementById("goalStarting").value = 0;

    refresh();

  });


// -------------------------
// ADD TRANSACTION
// -------------------------

document
  .getElementById("moneyForm")
  .addEventListener("submit", function(e) {

    e.preventDefault();

    const amount =
      Number(document.getElementById("moneyAmount").value);

    const type =
      document.getElementById("moneyType").value;

    const goalId =
      document.getElementById("moneyGoal").value;

    const note =
      document.getElementById("moneyNote").value.trim();

    if (amount <= 0) {

      alert("Enter a valid amount.");

      return;

    }


    const transaction = {

      id:
        "transaction_" +
        Date.now(),

      date: today(),

      amount,

      type,

      goalId,

      note

    };


    data.transactions.push(transaction);


    // Add money to selected goal
    if (
      type === "income" ||
      type === "allocation"
    ) {

      const goal =
        data.goals.find(g => g.id === goalId);

      if (goal) {
        goal.amount += amount;
      }

    }


    // Spending removes money from selected goal
    if (type === "expense") {

      const goal =
        data.goals.find(g => g.id === goalId);

      if (goal) {
        goal.amount -= amount;

        if (goal.amount < 0) {
          goal.amount = 0;
        }
      }

    }


    saveData();

    this.reset();

    refresh();

  });


// -------------------------
// DEPOP
// -------------------------

document
  .getElementById("depopForm")
  .addEventListener("submit", function(e) {

    e.preventDefault();

    const item =
      document.getElementById("depopItem").value.trim();

    const sale =
      Number(document.getElementById("depopSale").value);

    const cost =
      Number(document.getElementById("depopCost").value) || 0;

    const fees =
      Number(document.getElementById("depopFees").value) || 0;

    const source =
      document.getElementById("depopSource").value.trim();


    if (!item || sale <= 0) {

      alert("Enter an item and sale amount.");

      return;

    }


    const profit =
      sale - cost - fees;


    data.depop.push({

      id:
        "depop_" +
        Date.now(),

      date: today(),

      item,

      sale,

      cost,

      fees,

      source,

      profit

    });


    saveData();

    this.reset();

    document.getElementById("depopCost").value = 0;
    document.getElementById("depopFees").value = 0;

    refresh();

  });


// -------------------------
// DELETE
// -------------------------

function deleteActivity(id, isDepop) {

  if (!confirm("Delete this entry?")) {
    return;
  }


  if (isDepop) {

    data.depop =
      data.depop.filter(x => x.id !== id);

  } else {

    const transaction =
      data.transactions.find(x => x.id === id);

    if (transaction) {

      const goal =
        data.goals.find(g =>
          g.id === transaction.goalId
        );

      if (goal) {

        if (
          transaction.type === "income" ||
          transaction.type === "allocation"
        ) {

          goal.amount -= transaction.amount;

        }

        if (transaction.type === "expense") {

          goal.amount += transaction.amount;

        }

      }

    }


    data.transactions =
      data.transactions.filter(
        x => x.id !== id
      );

  }


  saveData();

  refresh();

}


// -------------------------
// EXPORT

document
  .getElementById("exportBtn")
  .addEventListener("click", function() {

    const blob =
      new Blob(
        [JSON.stringify(data, null, 2)],
        {type: "application/json"}
      );

    const url =
      URL.createObjectURL(blob);

    const a =
      document.createElement("a");

    a.href = url;

    a.download =
      "moneytrack-backup.json";

    a.click();

    URL.revokeObjectURL(url);

  });


// -------------------------
// IMPORT
// -------------------------

document
  .getElementById("importBtn")
  .addEventListener("click", function() {

    document
      .getElementById("importFile")
      .click();

  });


document
  .getElementById("importFile")
  .addEventListener("change", function(e) {

    const file = e.target.files[0];

    if (!file) return;

    const reader =
      new FileReader();

    reader.onload = function(event) {

      try {

        const imported =
          JSON.parse(event.target.result);

        if (
          !imported.goals ||
          !imported.transactions ||
          !imported.depop
        ) {

          throw new Error();

        }

        data = imported;

        saveData();

        refresh();

        alert("Backup imported successfully.");

      } catch {

        alert("That file isn't a valid MoneyTrack backup.");

      }

    };

    reader.readAsText(file);

  });


// -------------------------
// RESET
// -------------------------

document
  .getElementById("resetBtn")
  .addEventListener("click", function() {

    const confirmReset =
      confirm(
        "Are you sure? This will delete ALL your MoneyTrack data."
      );

    if (!confirmReset) return;

    localStorage.removeItem(STORAGE_KEY);

    location.reload();

  });


// -------------------------
// REFRESH
// -------------------------

function refresh() {

  renderGoals();

  renderGoalSelect();

  updateStats();

  renderActivity();

}


// -------------------------
// START
// -------------------------

loadData();

refresh();

</script>
